## 4. `D4.9` — reconciliação de tolerância ZERO, REPRODUZIDA

Pareamento: `metrics.create_time` (início do bucket) contra `markPriceKlines.open_time`.
Comparação em `Decimal` quantizado a **8 casas**, `ROUND_HALF_UP`. Tradução de caminho por
`data/MANIFEST.md`; `data/` é gitignored, então o script roda a partir da raiz do repositório
principal e **não** entra na suíte (um teste que dependesse de `data/` nasceria vermelho em todo
worktree e em todo clone limpo).

```python
# python3 este_script.py   — a partir de /home/stharley/Documentos/projects/cripto-strategy
import csv
from decimal import Decimal, ROUND_HALF_UP
from datetime import datetime, timezone

Q8 = Decimal("0.00000001")

def load_metrics(path):
    out = {}
    with open(path, newline="") as fh:
        for row in csv.DictReader(fh):
            dt = datetime.strptime(row["create_time"], "%Y-%m-%d %H:%M:%S").replace(tzinfo=timezone.utc)
            out[int(dt.timestamp() * 1000)] = (row["sum_open_interest_value"], row["sum_open_interest"])
    return out

def load_mark(path):
    with open(path, newline="") as fh:
        return {int(r["open_time"]): r["close"] for r in csv.DictReader(fh)}

def recon(mpath, kpath):
    m, k = load_metrics(mpath), load_mark(kpath)
    hits = n = 0
    worst = Decimal(0)
    for t in sorted(set(m) & set(k)):
        oiv, oi = Decimal(m[t][0]), Decimal(m[t][1])
        if oi == 0:
            continue
        derived = (oiv / oi).quantize(Q8, rounding=ROUND_HALF_UP)
        close = Decimal(k[t]).quantize(Q8, rounding=ROUND_HALF_UP)
        n += 1
        if derived == close:
            hits += 1
        else:
            worst = max(worst, abs(derived - close) / close * Decimal(10000))
    return hits, n, worst

R = "data/binance"
for name, mp, kp in [
    ("BTCUSDT 2026-08-21", f"{R}/metrics/btcusdt/2026-08-21.csv", f"{R}/klines/g3/mp_btc_0821/BTCUSDT-5m-2026-08-21.csv"),
    ("BTCUSDT 2026-08-23", f"{R}/metrics/btcusdt/2026-08-23.csv", f"{R}/klines/g3/markPriceKlines/BTCUSDT-5m-2026-08-23.csv"),
    ("COTIUSDT 2026-08-23", f"{R}/metrics/alts/COTIUSDT-metrics-2026-08-23.csv", f"{R}/klines/g3/mp_COTIUSDT/COTIUSDT-5m-2026-08-23.csv"),
    ("DOGEUSDT 2026-08-23", f"{R}/metrics/alts/DOGEUSDT-metrics-2026-08-23.csv", f"{R}/klines/g3/mp_DOGEUSDT/DOGEUSDT-5m-2026-08-23.csv"),
    ("SLXUSDT 2026-08-23", f"{R}/metrics/alts/SLXUSDT-metrics-2026-08-23.csv", f"{R}/klines/g3/mp_SLXUSDT/SLXUSDT-5m-2026-08-23.csv"),
]:
    h, n, w = recon(mp, kp)
    print(f"{name:22s} {h}/{n} exatos a 8 casas  pior residuo={w:.4f} bp")
```
