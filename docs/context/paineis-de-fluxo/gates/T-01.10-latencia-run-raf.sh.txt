#!/usr/bin/env bash
# F-7 rAF (ADR-044 emenda F-7): companion of T-01.1 §4, 5 rounds; each round = busy0 then busy20.
set -u
WT=/home/stharley/Documentos/projects/cripto-strategy/.claude/worktrees/wf_820bac26-f99-1
S=/tmp/claude-1002/-home-stharley-Documentos-projects-cripto-strategy/6624939a-3712-4593-98e1-345f9c754d91/scratchpad
OUT="$1"; ROUNDS="${2:-5}"
STATE_DIR="$(cat "$S/state_dir")"
mkdir -p "$OUT"; cd "$WT"
cp "$S/91-raf.spec.ts" frontend/e2e/91-baseline-t011.spec.ts
BASE="$(cat "$STATE_DIR/base_url")"
run() { # run <tag> <busy_ms> <round>
  printf '%s %s uptime: %s\n' "$(date +%T)" "$1" "$(uptime)" >> "$OUT/uptime.log"
  BASELINE_ROUND="$3" BASELINE_PAN_BUSY_MS="$2" E2E_BASE_URL="$BASE" E2E_FACTS_FILE="$OUT/$1.jsonl" E2E_SHOTS_DIR="$OUT/shots" \
    frontend/node_modules/.bin/playwright test --config=frontend/playwright.config.ts e2e/91-baseline-t011 \
    > "$OUT/$1.log" 2>&1
  local rc=$?
  printf '%s rc=%s\n' "$1" "$rc" >> "$OUT/rc.log"
}
for r in $(seq 1 "$ROUNDS"); do
  run "r$r-busy0" 0 "$r"
  run "r$r-busy20" 20 "$r"
done
rm -f frontend/e2e/91-baseline-t011.spec.ts
echo done >> "$OUT/rc.log"
