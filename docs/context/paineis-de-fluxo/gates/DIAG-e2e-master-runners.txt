##### ro_proxy.py (proxy GET/HEAD/OPTIONS, recusa o resto)
"""Read-only pass-through proxy (GET/HEAD/OPTIONS only) to the real API. Diagnostic, never committed."""
import http.server
import sys
import urllib.error
import urllib.request

UPSTREAM = sys.argv[2] if len(sys.argv) > 2 else "http://127.0.0.1:8000"
PORT = int(sys.argv[1])
COUNTS = {"get": 0, "refused": 0}


class H(http.server.BaseHTTPRequestHandler):
    def _cors(self):
        self.send_header("Access-Control-Allow-Origin", "*")
        self.send_header("Access-Control-Allow-Methods", "GET, HEAD, OPTIONS")
        self.send_header("Access-Control-Allow-Headers", "*")

    def do_OPTIONS(self):
        self.send_response(204)
        self._cors()
        self.end_headers()

    def _forward(self, head=False):
        COUNTS["get"] += 1
        try:
            with urllib.request.urlopen(UPSTREAM + self.path, timeout=60) as r:
                body = r.read()
                self.send_response(r.status)
                for k, v in r.headers.items():
                    if k.lower() in ("content-type", "cache-control"):
                        self.send_header(k, v)
        except urllib.error.HTTPError as e:
            body = e.read()
            self.send_response(e.code)
            self.send_header("Content-Type", e.headers.get("Content-Type", "text/plain"))
        self._cors()
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        if not head:
            self.wfile.write(body)

    def do_GET(self):
        self._forward()

    def do_HEAD(self):
        self._forward(head=True)

    def _refuse(self):
        COUNTS["refused"] += 1
        sys.stderr.write(f"REFUSED {self.command} {self.path}\n")
        self.send_response(405)
        self.end_headers()

    do_POST = do_PUT = do_PATCH = do_DELETE = _refuse

    def log_message(self, fmt, *args):
        sys.stderr.write("%s\n" % (fmt % args))


http.server.ThreadingHTTPServer(("127.0.0.1", PORT), H).serve_forever()
