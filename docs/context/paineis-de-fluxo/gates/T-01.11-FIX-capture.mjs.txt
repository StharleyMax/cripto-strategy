import { chromium } from "/home/stharley/Documentos/projects/cripto-strategy/.claude/worktrees/wf_b15536f1-3fb-7/frontend/node_modules/playwright/index.mjs";
const OUT = process.argv[2];
const TAG = process.argv[3];
const browser = await chromium.launch();
const result = {};
for (const vp of [{ width: 1280, height: 1200 }, { width: 1280, height: 800 }]) {
  const page = await browser.newPage({ viewport: vp, deviceScaleFactor: 1 });
  await page.goto("http://127.0.0.1:4321/symbol/BTCUSDT", { waitUntil: "load", timeout: 180000 });
  await page.locator(".tv-lightweight-charts").first().waitFor({ timeout: 180000 });
  await page.waitForTimeout(5000);
  const r = await page.evaluate(() => {
    const chart = window.__dbgChart;
    const cvdPane = chart.panes()[5];
    const delta = cvdPane.getSeries()[0];
    const served = delta.data().map((d) => d.value).filter((v) => v !== undefined);
    const modes = Object.fromEntries(Array.from(document.querySelectorAll("[data-right-scale-mode]")).map((e) => [e.dataset.testid, e.dataset.rightScaleMode]));
    const axisInk = ["liquidation-cohort-long", "liquidation-cohort-short", "oi-pane", "cvd-pane"].map((id) => {
      const cell = document.querySelector(`[data-testid="${id}"]`)?.closest("tr")?.lastElementChild;
      let ink = 0;
      for (const c of cell ? cell.querySelectorAll("canvas") : []) {
        const d = c.getContext("2d").getImageData(0, 0, c.width, c.height).data;
        for (let i = 0; i < d.length; i += 4) if (d[i + 3] && Math.abs(d[i] - 0x8b) <= 12 && Math.abs(d[i + 1] - 0x94) <= 12 && Math.abs(d[i + 2] - 0x9e) <= 12) ink++;
      }
      return [id, ink];
    });


    return {
      modes,
      cvd: { top: delta.coordinateToPrice(0), bottom: delta.coordinateToPrice(cvdPane.getHeight() - 1), min: Math.min(...served), max: Math.max(...served), n: served.length },
      axisInk: Object.fromEntries(axisInk),

      logoInChart: document.querySelectorAll('[data-testid="symbol-chart-host"] a[href*="tradingview"]').length,
      footer: document.querySelector('[data-testid="chart-attribution"] a')?.href ?? null,
      chromeLeft: Math.round(document.querySelector('[data-fact="chrome_mode:as_of"]').getBoundingClientRect().left + parseFloat(getComputedStyle(document.querySelector('[data-fact="chrome_mode:as_of"]')).paddingLeft)),
      marksKey: document.querySelector("[data-liquidation-marks-key]")?.textContent.trim() ?? null,
      restLegends: Array.from(document.querySelectorAll("[data-legend-value]")).map((e) => `${e.dataset.legendValue}=${e.textContent.trim()}`),
    };
  });
  await page.screenshot({ path: `${OUT}/T-01.11-FIX-${TAG}-${vp.width}x${vp.height}.png` });
  if (vp.height === 1200) {
    const box = await page.locator('[data-testid="price-pane"]').boundingBox();
    await page.mouse.move(box.x + box.width * 0.55, box.y + box.height * 0.5);
    await page.waitForTimeout(800);
    r.hoverLegends = await page.evaluate(() => Array.from(document.querySelectorAll("[data-legend-value]")).map((e) => `${e.dataset.legendValue}=${e.textContent.trim()}|${e.dataset.legendAbsence}`));
    await page.screenshot({ path: `${OUT}/T-01.11-FIX-${TAG}-1280x1200-crosshair.png` });
  }
  result[`${vp.width}x${vp.height}`] = r;
  await page.close();
}
await browser.close();
console.log(JSON.stringify(result, null, 1));
