# Validator's independent re-derivation of facts.txt blocks 1-4 (the designer committed no script for them).
import json, math, statistics as st, sys
V = sys.argv[1]; BAND = 335 * 0.20
def rows(n, tf): return {r["event_time"]: r for r in json.load(open(f"{V}/data/{n}-{tf}.json"))["rows"]}
for tf in ["1m", "5m", "15m", "1h", "4h"]:
    vol, op, cl = rows("vol", tf), rows("open", tf), rows("close", tf)
    pres = {t: float(r["value"]) for t, r in vol.items() if r["value"] is not None and r.get("absence") is None}
    partial = sum(1 for t, r in vol.items() if t in pres and r.get("coverage") and r["coverage"]["present"] < r["coverage"]["expected"])
    vs = sorted(pres.values()); mx = vs[-1]; p50 = st.median(vs)
    lin = {t: v / mx * BAND for t, v in pres.items()}
    lg = {t: max(0.0, math.log10(v) / math.log10(mx) * BAND) if v > 0 else 0 for t, v in pres.items()}
    up = {t: float(cl[t]["value"]) >= float(op[t]["value"]) for t in pres if t in op and t in cl and op[t]["value"] and cl[t]["value"]}
    ts = [t for t in pres if t in up]
    truth = sum(pres[t] for t in ts if up[t]) / sum(pres[t] for t in ts)
    share = lambda h: sum(h[t] for t in ts if up[t]) / sum(h[t] for t in ts)
    cnt = sum(up[t] for t in ts) / len(ts)
    below_abs = sum(1 for h in lin.values() if h < 1.70); below_zero = sum(1 for h in lin.values() if h < 5.10)
    print(f"{tf}: rows={len(vol)} present={len(pres)} partial={partial} max/p50={mx/p50:.1f}x lin_p50={st.median(lin.values()):.2f}px "
          f"lin<1px={sum(1 for h in lin.values() if h<1)} log_p50={st.median(lg.values()):.1f}px log_max/p50={BAND/st.median(lg.values()):.2f}x | "
          f"n_dir={len(ts)} upcount={cnt:.3f} truth={truth:.3f} lin={share(lin):.3f} log={share(lg):.3f} err_log={share(lg)-truth:+.3f} | "
          f"lin<absence={below_abs} lin<zero={below_zero}")
