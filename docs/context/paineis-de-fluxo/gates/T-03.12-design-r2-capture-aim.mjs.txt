// T-03.12 r2 capture: same rig as T-03.12-design-capture.mjs.txt (real app :4322 over the read-only proxy :8823),
// but it AIMS by the time the app itself reports under the crosshair (data-legend-bucket-ms of the OI legend),
// not by (target - windowStart)/step, which drifts when the canonical grid is not uniform in time.
// Mouse only (drag + wheel), no hook into the page, no writes. Usage: node capture2.mjs <out> <sym> <tf> <iso> <bars> <tag>
import { chromium } from "/home/stharley/Documentos/projects/cripto-strategy/.claude/worktrees/t-03-12/frontend/node_modules/playwright/index.mjs";
import fs from "node:fs";
const [OUT, SYMBOL, INTERVAL, TARGET_ISO, SPAN_RAW, TAG] = process.argv.slice(2);
const TARGET = Date.parse(TARGET_ISO), SPAN = Number(SPAN_RAW);
const STEP = { "1m": 60_000, "5m": 300_000, "15m": 900_000 }[INTERVAL];
const browser = await chromium.launch();
const page = await browser.newPage({ viewport: { width: 1600, height: 1200 }, deviceScaleFactor: 1 });
await page.goto(`http://127.0.0.1:4322/symbol/${SYMBOL}?interval=${INTERVAL}`, { waitUntil: "load", timeout: 180_000 });
await page.locator(".tv-lightweight-charts").first().waitFor({ timeout: 180_000 });
await page.waitForTimeout(4000);
const box = async () => page.locator('[data-testid="oi-pane"]').boundingBox();
const read = async () => page.evaluate(() => {
  const v = document.querySelector('[data-legend-value="oi"]'); const o = document.querySelector('[data-legend-ohlc="oi"]');
  return { ms: Number(v?.getAttribute("data-legend-bucket-ms") || NaN), df: o?.getAttribute("data-legend-derived-from") || "",
           samples: o?.getAttribute("data-legend-samples") || "", text: (o?.textContent || "").replace(/\s+/g, " ").trim().slice(0, 110) };
});
const at = async (x) => { const b = await box(); await page.mouse.move(x, b.y + b.height * 0.6, { steps: 2 }); await page.waitForTimeout(150); return read(); };
const calib = async () => { const b = await box(); const x1 = b.x + b.width * 0.2, x2 = b.x + b.width * 0.7;
  const r1 = await at(x1), r2 = await at(x2); return { b, x1, t1: r1.ms, mpp: (r2.ms - r1.ms) / (x2 - x1) }; };
async function aim(iter) {
  for (let i = 0; i < iter; i++) {
    const c = await calib(); if (!Number.isFinite(c.mpp) || c.mpp <= 0) { await page.waitForTimeout(800); continue; }
    const want = c.b.x + c.b.width * 0.55; const tWant = c.t1 + (want - c.x1) * c.mpp;
    const dx = (tWant - TARGET) / c.mpp; if (Math.abs(dx) < 3) return;
    const cl = Math.max(-c.b.width * 0.4, Math.min(c.b.width * 0.4, dx)); const y = c.b.y + c.b.height * 0.5;
    await page.mouse.move(c.b.x + c.b.width * 0.5, y); await page.mouse.down();
    await page.mouse.move(c.b.x + c.b.width * 0.5 + cl, y, { steps: 8 }); await page.mouse.up(); await page.waitForTimeout(1200);
  }
}
await aim(25);
for (let i = 0; i < 120; i++) {
  const c = await calib(); const bars = ((c.b.width - 70) * c.mpp) / STEP;
  if (!Number.isFinite(bars) || bars <= 0) { await page.mouse.move(c.b.x + c.b.width * 0.5, c.b.y + c.b.height * 0.5); await page.mouse.wheel(0, 120); await page.waitForTimeout(300); continue; }
  if (Math.abs(bars - SPAN) / SPAN < 0.15) break;
  const xT = Math.max(c.b.x + 5, Math.min(c.b.x + c.b.width - 80, c.x1 + (TARGET - c.t1) / c.mpp)); await page.mouse.move(xT, c.b.y + c.b.height * 0.5);
  await page.mouse.wheel(0, bars > SPAN ? -120 : 120); await page.waitForTimeout(150);
}
await aim(25);
// sweep: one hover per bar across the plot, so every column of the screenshot has its served fact
const c = await calib(); const sweep = [];
for (let x = c.b.x + 4; x < c.b.x + c.b.width - 72; x += Math.max(2, STEP / c.mpp)) { const r = await at(x); sweep.push({ x: Math.round(x), ...r }); }
await page.mouse.move(2, 2); await page.waitForTimeout(800);
const b = await box();
await page.screenshot({ path: `${OUT}/${TAG}-oi.png`, clip: { x: 0, y: b.y - 4, width: 1600, height: b.height + 8 } });
const iso = (ms) => (Number.isFinite(ms) ? new Date(ms).toISOString().slice(5, 16) + "Z" : "");
const trans = sweep.filter((s, i) => i && (s.df !== sweep[i - 1].df)).map((s) => ({ x: s.x, at: iso(s.ms), df: s.df || "(ausente)" }));
fs.writeFileSync(`${OUT}/${TAG}-facts.json`, JSON.stringify({ SYMBOL, INTERVAL, TARGET_ISO, bars: ((c.b.width - 70) * c.mpp) / STEP, first: iso(sweep[0]?.ms), last: iso(sweep.at(-1)?.ms), transitions: trans, sweep: sweep.map((s) => ({ ...s, ms: iso(s.ms) })) }, null, 1));
console.log(JSON.stringify({ tag: TAG, bars: Math.round(((c.b.width - 70) * c.mpp) / STEP), first: iso(sweep[0]?.ms), last: iso(sweep.at(-1)?.ms), transitions: trans.slice(0, 12) }));
await browser.close();
