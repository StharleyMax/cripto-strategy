# T-03.12 quant-architect: read-only fetch of the 8 OI series from the production API, THROUGH the
# read-only proxy of gates/T-01.11-r2-proxy.mjs.txt (GET/HEAD only, 405 otherwise) on :8822.
# Run the proxy first:  node T-01.11-r2-proxy.mjs  (it prints proxy_get_requests_total / refused).
# Output prod-oi.json is raw third-party data and is NOT versioned (CLAUDE.md "Dado bruto").
# series_key_id computed by the worktree's own builders:
#   binance_open_interest_key(instrument_id=s).series_key_id()        -> hist 5m
#   binance_open_interest_poll_key(instrument_id=s).series_key_id()   -> poll 1m
import json,time,urllib.request
now=int(time.time()*1000)//60000*60000
ids={'BTCUSDT':('94c3d3dd5f45abcb801a53e4a8b52ea81ea2479a9cdd51d90cd2cb6895e1a4a9','d28b119002a9dfb2f5b99b1ad4bf29f9746591200fc21dfea501914bfa535897'),
'ETHUSDT':('e627e6554365b490e3d35df47c8a7038b4b41d0724e41d73dcfc17fbac32741a','d6566b6ba09b7fe5e85bf94fdc09c66e2e7f14e26eb373adedc8da0854f3509c'),
'LINKUSDT':('6f4a8efae9e4a61b4ac469e9e6ecf3624ef43e0a65355005aed73e9cda7d8ff0','52885576cb4a95d5cb5ed84114b968ae97f647828cca358f60c1bf00bc1943f3'),
'SOLUSDT':('e881ab713adbae083e92dfe0bf705e1a53aaa9b9952e9aca0cede876ce127f1b','e9917adf8ce5e3d7c08b443fdbdcddfe95b373a351c9061ca4848907f6da31e8')}
start=1790294400000  # 2026-09-25T00:00Z
out={'knowledge_time_ms':now,'window_start_ms':start}
for s,(h,p) in ids.items():
  for name,sid,iv in (('hist',h,'5m'),('poll',p,'1m')):
    url=f'http://127.0.0.1:8822/api/v1/series-history?series_key_id={sid}&symbol={s}&interval={iv}&window_start_ms={start}&window_end_ms={now}&knowledge_time_ms={now}&bar_policy=final_only'
    out[f'{s}:{name}']=json.load(urllib.request.urlopen(url,timeout=30))
json.dump(out,open('prod-oi.json','w'))
