// T-03.12 design_gate capture: real app (next start :4322 of the worktree) over real data
// (T-03.12-design-proxy.py.txt on :8823: OI series-history from the worktree's own route
// function over a read-only md.series export; everything else passed through to prod :8000).
// Pans/zooms the ONE chart with mouse drag + wheel (no hook into the page), then screenshots
// the OI pane around each measured regime event and reads the pane's own data-* facts.
// No writes anywhere. Usage: node capture.mjs <out-dir> <symbol> <interval> <targetIso> <spanBars> <tag>
import { chromium } from "/home/stharley/Documentos/projects/cripto-strategy/.claude/worktrees/t-03-12/frontend/node_modules/playwright/index.mjs";
import fs from "node:fs";

const [OUT, SYMBOL, INTERVAL, TARGET_ISO, SPAN_RAW, TAG] = process.argv.slice(2);
const BASE = "http://127.0.0.1:4322";
const SPAN = Number(SPAN_RAW);
const STEP_MS = 60_000; // the canonical grid is 1m in every TF (S-1): logical index = 1m slot
const browser = await chromium.launch();
const page = await browser.newPage({ viewport: { width: 1600, height: 1200 }, deviceScaleFactor: 1 });
const errors = [];
page.on("console", (m) => { if (m.type() === "error") errors.push(m.text().slice(0, 200)); });
await page.goto(`${BASE}/symbol/${SYMBOL}?interval=${INTERVAL}`, { waitUntil: "load", timeout: 180_000 });
await page.locator(".tv-lightweight-charts").first().waitFor({ timeout: 180_000 });
await page.waitForTimeout(4000);

const host = page.locator('[data-testid="symbol-chart-host"], [data-visible-logical-from]').first();
const range = async () => ({
  from: Number(await host.getAttribute("data-visible-logical-from")),
  to: Number(await host.getAttribute("data-visible-logical-to")),
});
const windowStartMs = Number(await page.locator("main[data-window-start-ms]").getAttribute("data-window-start-ms"));
let target = (Date.parse(TARGET_ISO) - windowStartMs) / STEP_MS;
// a prepended older page moves logical 0, so the target is re-read off the CURRENT window each time
const retarget = async () => { target = (Date.parse(TARGET_ISO) - Number(await page.locator("main[data-window-start-ms]").getAttribute("data-window-start-ms"))) / STEP_MS; };
const oiBox = async () => (await page.locator('[data-testid="oi-pane"]').boundingBox());

const xOf = (r, b, logical) => b.x + (logical - r.from) * ((b.width) / (r.to - r.from));
async function pan(maxIter) {
  for (let i = 0; i < maxIter; i++) {
    await retarget();
    const r = await range();
    const want = r.from + (r.to - r.from) * 0.6;
    const b = await oiBox();
    const dx = (want - target) * ((b.width) / (r.to - r.from));
    if (Math.abs(dx) < 4) break;
    const clamped = Math.max(-b.width * 0.4, Math.min(b.width * 0.4, dx));
    const y = b.y + b.height * 0.5;
    await page.mouse.move(b.x + b.width * 0.5, y);
    await page.mouse.down();
    await page.mouse.move(b.x + b.width * 0.5 + clamped, y, { steps: 8 });
    await page.mouse.up();
    await page.waitForTimeout(1200); // the pager may fetch an older page
  }
}
await pan(20);
// zoom around the target (the library zooms around the cursor)
for (let i = 0; i < 160; i++) {
  await retarget();
  const r = await range();
  const span = r.to - r.from;
  if (Math.abs(span - SPAN) / SPAN < 0.12) break;
  const b = await oiBox();
  await page.mouse.move(xOf(r, b, target), b.y + b.height * 0.5);
  await page.mouse.wheel(0, span > SPAN ? -120 : 120);
  await page.waitForTimeout(120);
}
await pan(20);
await page.mouse.move(2, 2);
await page.waitForTimeout(1500);
await retarget();
const r = await range();
const b = await oiBox();
const facts = {
  symbol: SYMBOL, interval: INTERVAL, target_iso: TARGET_ISO, target_logical: target, visible: r, windowStartMs: Number(await page.locator("main[data-window-start-ms]").getAttribute("data-window-start-ms")),
  oi_pane_box: b, console_errors: errors,
  oi_attrs: await page.evaluate(() => {
    const el = document.querySelector('[data-testid="oi-pane"]');
    const a = {}; for (const x of el.attributes) if (x.name.startsWith("data-") && x.value.length < 80) a[x.name] = x.value;
    return a;
  }),
  legend_at_rest: await page.evaluate(() => {
    const el = document.querySelector('[data-legend-ohlc="oi"]');
    return el ? { text: el.textContent.replace(/\s+/g, " ").trim(), df: el.getAttribute("data-legend-derived-from"), samples: el.getAttribute("data-legend-samples") } : null;
  }),
};
// hover probes at the target +/- a few bars: legend under the crosshair
facts.hover = [];
const pxPerBar = (b.width) / (r.to - r.from);
for (const off of [-30, -15, -1, 0, 1, 15, 30]) {
  const x = b.x + (target + off - r.from) * pxPerBar;
  await page.mouse.move(x, b.y + b.height * 0.6, { steps: 3 });
  await page.waitForTimeout(250);
  facts.hover.push({ off, legend: await page.evaluate(() => {
    const el = document.querySelector('[data-legend-ohlc="oi"]');
    return el ? { text: el.textContent.replace(/\s+/g, " ").trim().slice(0, 140), df: el.getAttribute("data-legend-derived-from"), samples: el.getAttribute("data-legend-samples") } : null;
  }) });
}
await page.mouse.move(2, 2);
await page.waitForTimeout(500);
const priceBox = await page.locator('[data-testid="price-pane"]').boundingBox();
await page.screenshot({ path: `${OUT}/${TAG}-oi.png`, clip: { x: 0, y: b.y - 4, width: 1600, height: b.height + 8 } });
await page.screenshot({ path: `${OUT}/${TAG}-price+oi.png`, clip: { x: 0, y: priceBox.y, width: 1600, height: b.y + b.height - priceBox.y + 4 } });
fs.writeFileSync(`${OUT}/${TAG}-facts.json`, JSON.stringify(facts, null, 1));
console.log(JSON.stringify({ tag: TAG, visible: r, target, errors: errors.length, legend: facts.legend_at_rest }));
await browser.close();
